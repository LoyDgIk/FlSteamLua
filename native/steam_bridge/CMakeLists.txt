cmake_minimum_required(VERSION 3.20)
project(steam_bridge LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "steam_bridge requires Windows x64")
endif()
add_library(steam_bridge SHARED steam_bridge.cpp)
target_compile_features(steam_bridge PRIVATE cxx_std_17)
target_compile_definitions(steam_bridge PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(steam_bridge PRIVATE /W4 /permissive- /EHsc)
endif()
set_target_properties(steam_bridge PROPERTIES OUTPUT_NAME steam_bridge)
